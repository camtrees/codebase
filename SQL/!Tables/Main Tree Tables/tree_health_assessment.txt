-- public.tree_health_assessment definition

-- Drop table

-- DROP TABLE public.tree_health_assessment;

CREATE TABLE public.tree_health_assessment (
	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL, -- The PostGreSQL unique primary key (generated when the record is created)
	epicollect_record_type_id int4 NOT NULL, -- The EpiCollect record_type (pathway through the app) used to collect the data
	"date" date NOT NULL, -- The date the health assessment was taken
	volunteer_id int4 NOT NULL, -- The volunteer who performed the health assessment
	health_id int4 NULL, -- The health of the tree
	height_in_inches numeric(6, 2) NULL, -- The height of the tree
	diameter_in_inches numeric(6, 2) NULL, -- The diameter of the tree
	blight bool NULL, -- Does the tree have blight?
	stump_sprouting bool NULL, -- Is there any stump sprouting?
	catkins bool NULL, -- Does the tree have catkins?
	blossoms bool NULL, -- Does the tree have blossoms?
	nuts_id int4 NULL, -- Are there Nuts (Burrs) Present?
	note text NULL, -- Any notes provided by the volunteer when performing the tree health assessment
	date_created date NOT NULL, -- The date the record was created
	date_updated date NOT NULL, -- The date the record was most recently updated
	tree_id int4 NOT NULL, -- The tree for which the health assessment was taken
	CONSTRAINT tree_health_assesment_date_not_null NOT NULL date,
	CONSTRAINT tree_health_assesment_epicollect_record_type_id_not_null NOT NULL epicollect_record_type_id,
	CONSTRAINT tree_health_assesment_id_not_null NOT NULL id,
	CONSTRAINT tree_health_assesment_pkey PRIMARY KEY (id),
	CONSTRAINT tree_health_assesment_volunteer_id_not_null NOT NULL volunteer_id,
	CONSTRAINT tree_health_assessment_date_created_not_null NOT NULL date_created,
	CONSTRAINT tree_health_assessment_date_updated_not_null NOT NULL date_updated,
	CONSTRAINT tree_health_assessment_tree_id_not_null NOT NULL tree_id
);
COMMENT ON TABLE public.tree_health_assessment IS 'Tree measurements and observations to access the health of the tree';
