-- public.tree definition

-- Drop table

-- DROP TABLE public.tree;

CREATE TABLE public.tree (
	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL, -- The PostGreSQL unique primary key (generated when the record is created)
	epicollect_record_type_id int4 NOT NULL, -- The EpiCollect record_type (pathway through the app) used to collect the data
	site_id int4 NOT NULL, -- The site at which the tree was planted
	"number" text NOT NULL, -- The tree number as text. CAM planted trees are integers with leading zeroes. WildCAM trees use the longitude and latitude as a tree number. WildTACF trees use a tree id in the format 'ME-XXnnn' where XX are two uppercase letters and nnn are three digits.
	"date" date NOT NULL, -- The date the tree was planted OR for Wild trees the date the tree was observed
	volunteer_id int4 NOT NULL, -- The volunteer who planted the tree OR for Wild trees the volunteer who observed the tree
	primary_caretaker_id int4 NULL, -- The primary caretaker of the tree. Responsible for watering, etc.
	secondary_caretaker_id int4 NULL, -- The secondary caretaker of the tree. Responsible for watering, etc.
	longitude float8 NULL, -- The GPS longitude
	latitude float8 NULL, -- The GPS latitude
	geog public.geography(point, 4326) GENERATED ALWAYS AS (st_setsrid(st_makepoint(longitude, latitude), 4326)::geography) STORED NULL, -- An ArcGIS POINT value used to plot the tree on a map
	gps_locked bool DEFAULT false NOT NULL, -- 'true' means our Python scripts should NOT be allowed to update the Longitude and Latitude values
	form_id int4 NULL, -- The tree form
	planting_method_id int4 NULL, -- The method used to plant the tree
	wire_fence bool NULL, -- Was a wire fence placed around the tree when it was planted?
	tag_info text NULL, -- Additional info on the Mother and Father trees
	classification text NULL, -- The botanical categorization of the tree
	note text NULL, -- Any notes provided by the volunteer when PLANTING or OBSERVING the tree
	date_created date NOT NULL, -- The date the record was created
	date_updated date NOT NULL, -- The date the record was most recently updated
	elevation_in_feet float8 NULL, -- The elevation of the tree
	access_path_id int4 NULL, -- What kind of road, trail, etc is used to access the tree
	access_level_id int4 NULL, -- How easy is it to access the tree
	access_method_id int4 NULL, -- What method of access is necessary to pollinate or harvest the tree
	access_note text NULL, -- Comments regarding tree accessibility
	mother_tree_id int4 NULL, -- Mother tree
	mother_tree_other text NULL, -- Mother tree recorded during planting when the mother tree is not in the current tree_parent_type list
	father_tree_id int4 NULL, -- Father tree
	father_tree_other text NULL, -- Father tree recorded during planting when the father tree is not in the current tree_parent_type list
	parent_tree_note text NULL, -- Additional Mother/Father tree info provided when the tree was planted
	CONSTRAINT tree_date_created_not_null NOT NULL date_created,
	CONSTRAINT tree_date_not_null NOT NULL date,
	CONSTRAINT tree_date_updated_not_null NOT NULL date_updated,
	CONSTRAINT tree_epicollect_record_type_id_not_null NOT NULL epicollect_record_type_id,
	CONSTRAINT tree_gps_locked_not_null NOT NULL gps_locked,
	CONSTRAINT tree_id_not_null NOT NULL id,
	CONSTRAINT tree_number_not_null NOT NULL number,
	CONSTRAINT tree_pkey PRIMARY KEY (id),
	CONSTRAINT tree_site_id_not_null NOT NULL site_id,
	CONSTRAINT tree_volunteer_id_not_null NOT NULL volunteer_id
);
CREATE INDEX tree_geog_index ON public.tree USING gist (geog);
CREATE UNIQUE INDEX tree_site_id_and_number_unique ON public.tree USING btree (site_id, number);
COMMENT ON TABLE public.tree IS 'CAM planted trees or Wild trees under observation';
