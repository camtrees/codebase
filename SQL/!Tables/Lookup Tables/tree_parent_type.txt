	note text NULL, -- Any notes about the tree. Perhaps the tree's mother, or father, or location. Etc.
	CONSTRAINT tree_parent_type_id_not_null NOT NULL id,
	CONSTRAINT tree_parent_type_parent_not_null NOT NULL parent,
	CONSTRAINT tree_parent_type_pkey PRIMARY KEY (id)
);
