	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL, -- The PostGreSQL unique primary key (generated when the record is created)
	care_action text NOT NULL, -- The care action performed
	CONSTRAINT tree_care_action_type_care_action_not_null NOT NULL care_action,
	CONSTRAINT tree_care_action_type_id_not_null NOT NULL id,
	CONSTRAINT tree_care_action_type_pkey PRIMARY KEY (id)
);
CREATE UNIQUE INDEX tree_care_action_type_care_action_unique ON public.tree_care_action_type USING btree (care_action);
COMMENT ON TABLE public.tree_care_action_type IS 'Lookup Table - Care actions (water, weed, fertilize, prune)';
