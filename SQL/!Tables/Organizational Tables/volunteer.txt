-- public.volunteer definition

-- Drop table

-- DROP TABLE public.volunteer;

CREATE TABLE public.volunteer (
	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL, -- The PostGreSQL unique primary key (generated when the record is created)
	email text NOT NULL, -- The volunteer's email address
	first_name text NOT NULL -- The volunteer's first name,
	last_name text NOT NULL, -- The volunteer's last name
	title text NULL -- The volunteer's title,
	status_id int4 NOT NULL, -- The volunteer's status (active, inactive, etc)
	town_id int4 NULL -- The volunteer's home town,
	date_created date NOT NULL, -- The date the record was created
	date_updated date NOT NULL, -- The date the record was most recently updated
	CONSTRAINT volunteer_date_created_not_null NOT NULL date_created,
	CONSTRAINT volunteer_date_updated_not_null NOT NULL date_updated,
	CONSTRAINT volunteer_email_not_null NOT NULL email,
	CONSTRAINT volunteer_first_name_not_null NOT NULL first_name,
	CONSTRAINT volunteer_id_not_null NOT NULL id,
	CONSTRAINT volunteer_last_name_not_null NOT NULL last_name,
	CONSTRAINT volunteer_pkey PRIMARY KEY (id),
	CONSTRAINT volunteer_status_id_not_null NOT NULL status_id
);
CREATE UNIQUE INDEX volunteer_email_unique ON public.volunteer USING btree (email);
CREATE UNIQUE INDEX volunteer_first_name_and_last_name_unique ON public.volunteer USING btree (first_name, last_name);
COMMENT ON TABLE public.volunteer IS 'Volunteers. All persons referenced in any table must appear in this table.';

COMMENT ON COLUMN public.volunteer.id IS 'The PostGreSQL unique primary key (generated when the record is created)';

COMMENT ON COLUMN public.volunteer.title IS 'The volunteer''s title';
COMMENT ON COLUMN public.volunteer.status_id IS 'The volunteer''s status (active, inactive, etc)';
