-- public.site definition

-- Drop table

-- DROP TABLE public.site;

CREATE TABLE public.site (
	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL, -- The PostGreSQL unique primary key (generated when the record is created)
	"name" text NOT NULL, -- The Site name
	organization_id int4 NOT NULL, -- The CAM Org short code
	town_id int4 NULL, -- The town in which the site resides
	"location" text NULL, -- The address (if sensible) of the site
	location_note text NULL, -- Info useful to help locate the site
	volunteer_id int4 NULL, -- The site contact person
	date_created date NOT NULL, -- The date the record was created
	date_updated date NOT NULL, -- The date this record was most recently updated
	url text NULL, -- The URL of the site (if there is one)
	primary_caretaker_id int4 NULL, -- The primary volunteer who has accepted responsibility for monitoring and maintaining all the trees at a given site
	secondary_caretaker_id int4 NULL, -- The secondary volunteer who has accepted responsibility for monitoring and maintaining all the trees at a given site
	CONSTRAINT site_date_created_not_null NOT NULL date_created,
	CONSTRAINT site_date_updated_not_null NOT NULL date_updated,
	CONSTRAINT site_id_not_null NOT NULL id,
	CONSTRAINT site_name_not_null NOT NULL name,
	CONSTRAINT site_organization_id_not_null NOT NULL organization_id,
	CONSTRAINT site_pkey PRIMARY KEY (id)
);
CREATE UNIQUE INDEX site_name_unique ON public.site USING btree (name);
COMMENT ON TABLE public.site IS 'Sites where trees are planted';
