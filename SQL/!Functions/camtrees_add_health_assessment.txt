------------------------------------------------------------------------------------------
-- Function: camtrees_add_health_assessment
--
-- Purpose: Add a new tree_health_assessment from an EpiCollect ONE record
--
-- Returns: the id number of the health_assessment added
------------------------------------------------------------------------------------------
-- DROP FUNCTION public.camtrees_add_health_assessment(text, text, text, date, text, text, text, text, text, text, text, text, text, text);

CREATE OR REPLACE FUNCTION public.camtrees_add_health_assessment(
	p_epicollect_record_type  text,
	p_site_name               text,
	p_tree_number             text,
	p_date                    date,
	p_volunteer_email         text,
	p_health                  text,
	p_height_in_inches        text,
	p_diameter_in_inches      text,
	p_blight                  text,
	p_stump_sprouting         text,
	p_catkins                 text,
	p_blossoms                text,
	p_nuts                    text,
	p_note                    text
	)
	RETURNS  integer
	LANGUAGE plpgsql
	AS       $function$

	DECLARE
		v_epicollect_record_type_id INTEGER;
		v_site_id                   INTEGER;
		v_volunteer_id              INTEGER;
		v_health_id                 INTEGER;
		v_nuts_id                   INTEGER;
		v_tree_id                   INTEGER;

	BEGIN
		-- Get epicollect_record_type_id
		SELECT id INTO v_epicollect_record_type_id FROM epicollect_record_type WHERE record_type = lower(p_epicollect_record_type);

		-- Get site_id
		SELECT id INTO v_site_id FROM site WHERE name = p_site_name;
		IF v_site_id IS NULL THEN
			RAISE EXCEPTION 'Site not found: %', p_site_name;
		END IF;

		-- Get tree_id
		SELECT id INTO v_tree_id FROM tree WHERE site_id=v_Site_id and number=p_tree_number;
		IF v_tree_id IS NULL THEN
			RAISE EXCEPTION 'Tree not found. Site:"%" Tree_Number:"%"', p_site_name, p_tree_number;
		END IF;
		
		-- Get the volunteer_id
		SELECT id INTO v_volunteer_id FROM volunteer WHERE email = lower(p_volunteer_email);

		-- Get health_id if health provided
		IF p_health IS NOT NULL THEN
			SELECT id INTO v_health_id FROM tree_health_type WHERE health = lower(p_health);
		END IF;
		
		-- Get nuts_id if nuts provided
		IF p_nuts IS NOT NULL THEN
			SELECT id INTO v_nuts_id FROM tree_health_nuts_type WHERE nuts = lower(p_nuts);
		END IF;

		-- Replace 'NULL' string values with NULL for various input parameters
		p_height_in_inches   := NULLIF(p_height_in_inches   , 'NULL');
		p_diameter_in_inches := NULLIF(p_diameter_in_inches , 'NULL');
		p_blight             := NULLIF(p_blight             , 'NULL');
		p_stump_sprouting    := NULLIF(p_stump_sprouting    , 'NULL');
		p_catkins            := NULLIF(p_catkins            , 'NULL');
		p_blossoms           := NULLIF(p_blossoms           , 'NULL');
		p_note               := NULLIF(p_note               , 'NULL');
			
		-- Insert the new tree
		INSERT INTO tree_health_assessment (
			epicollect_record_type_id,
			tree_id,
			date,
			volunteer_id,
			health_id,
			height_in_inches,
			diameter_in_inches,
			blight,
			stump_sprouting,
			catkins,
			blossoms,
			nuts_id,
			note
			)
		VALUES (
			v_epicollect_record_type_id,
			v_tree_id,
			p_date,
			v_volunteer_id,
			v_health_id,
			p_height_in_inches::INTEGER,
			p_diameter_in_inches::NUMERIC(6,2),
			p_blight::BOOLEAN,
			p_stump_sprouting::BOOLEAN,
			p_catkins::BOOLEAN,
			p_blossoms::BOOLEAN,
		    v_nuts_id,
			p_note
			) RETURNING id INTO v_tree_id;
		
		RETURN v_tree_id;
	END;
	$function$
;
