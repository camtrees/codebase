------------------------------------------------------------------------------------------
-- Function: camtrees_update_access_method
--
-- Purpose: Update tree's access_method
--
-- Returns: true  - the tree's form was updated
--            false - the tree's form was NOT updated
------------------------------------------------------------------------------------------
-- DROP FUNCTION public.camtrees_update_access_method(text, text, text);

CREATE OR REPLACE FUNCTION public.camtrees_update_access_method(
	p_site_name text,
	p_tree_number text,
	p_access_method text
	)
	RETURNS  boolean
	LANGUAGE plpgsql
	AS       $function$

	DECLARE
		v_site_id          INTEGER;
		v_access_method_id INTEGER;
		v_tree_id          INTEGER;
		v_rows_updated     INTEGER;

	BEGIN
		-- Get the site id
		SELECT id INTO v_site_id
		FROM site
		WHERE name = trim(p_site_name);
		IF v_site_id IS NULL THEN
			RAISE EXCEPTION 'site not found: %', p_site_name;
		END IF;

		-- Replace 'NULL' string values with NULL for various input parameters
		p_access_method  := NULLIF(p_access_method , 'NULL');
		
		-- Get the access_method id
		IF p_access_method IS NOT NULL THEN
			SELECT id INTO v_access_method_id FROM tree_access_method_type WHERE access_method = lower(p_access_method);
		END IF;
		IF v_access_method_id IS NULL THEN
			RAISE EXCEPTION 'Access method not found: %', p_access_method;
		END IF;

		-- Get the tree id
		SELECT id INTO v_tree_id
		FROM tree
		WHERE site_id = v_site_id AND number = p_tree_number;
		IF v_tree_id IS NULL THEN
			RAISE EXCEPTION 'tree not found. Site: "%" Tree Number: "%"', p_site_name, p_tree_number;
		END IF;
		
		-- Update the tree with provided values
		UPDATE tree
		SET 
			access_method_id = v_access_method_id
		WHERE id = v_tree_id;
		
		GET DIAGNOSTICS v_rows_updated = ROW_COUNT;
		
		RETURN v_rows_updated > 0;
	END;
	$function$
;

-- Permissions

ALTER FUNCTION public.camtrees_update_access_method(text, text, text) OWNER TO neondb_owner;
GRANT ALL ON FUNCTION public.camtrees_update_access_method(text, text, text) TO neondb_owner;
