--DROP VIEW IF EXISTS cam_tree_care_action;

-- public.cam_tree_care_action source

CREATE OR REPLACE VIEW public.cam_tree_care_action
AS SELECT (v.first_name || ' '::text) || v.last_name AS volunteer,
    e.record_type,
    s.name AS site,
    t.number AS tree_number,
    tca.date,
    v.email AS volunteer_email,
    tcat.care_action,
    tca.date_created,
    tca.date_updated,
    tca.note,
    tca.tree_id
   FROM tree_care_action tca
     LEFT JOIN epicollect_record_type e ON tca.epicollect_record_type_id = e.id
     LEFT JOIN tree t ON tca.tree_id = t.id
     LEFT JOIN site s ON t.site_id = s.id
     LEFT JOIN volunteer v ON tca.volunteer_id = v.id
     LEFT JOIN tree_care_action_type tcat ON tca.care_action_id = tcat.id
  ORDER BY s.name, t.number, tca.date, tcat.care_action;

COMMENT ON VIEW public.cam_tree_care_action IS 'Care actions performed on trees';

-- Permissions

ALTER TABLE public.cam_tree_care_action OWNER TO neondb_owner;
GRANT ALL ON TABLE public.cam_tree_care_action TO neondb_owner;
