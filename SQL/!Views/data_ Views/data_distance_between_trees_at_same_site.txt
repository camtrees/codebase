------------------------------------------------------------------------------------------
-- View: data_distance_between_trees_at_same_site
--
-- Purpose: Show tree to tree distance but only for trees at the same site
--
-- Create Order :
--		MUST BE AFTER
--			site
--			tree
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_distance_between_trees_at_same_site;

-- public.data_distance_between_trees_at_same_site source

CREATE OR REPLACE VIEW public.data_distance_between_trees_at_same_site
AS SELECT s.name AS site,
    t1.tag_id AS tree1_tag_id,
    t2.tag_id AS tree2_tag_id,
    round(st_distance(t1.geog, t2.geog)) AS distance_in_meters,
    round((st_distance(t1.geog, t2.geog) * 3.280839895::double precision)::numeric, 1) AS distance_in_feet
   FROM tree t1
     JOIN tree t2 ON t1.site_id = t2.site_id AND t1.tag_id <> t2.tag_id
     JOIN site s ON t1.site_id = s.id
  WHERE st_distance(t1.geog, t2.geog) <> 0::double precision
  ORDER BY s.name, t1.tag_id, t2.tag_id;

COMMENT ON VIEW public.data_distance_between_trees_at_same_site IS 'Show tree to tree distance (in meters and feet) but only for trees at the same site';
