------------------------------------------------------------------------------------------
-- View: data_distance_trees_within_15_meters_of_each_other
--
-- Purpose: Show tree to tree distance for all trees within 15 meters of each other
--
-- Note: Efficient way using ST_DWithin to find tree distances
--
-- Create Order :
--		MUST BE AFTER
--			site
--			tree
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_distance_trees_within_15_meters_of_each_other;

-- public.data_distance_trees_within_15_meters_of_each_other source

CREATE OR REPLACE VIEW public.data_distance_trees_within_15_meters_of_each_other
AS SELECT s1.name AS tree1_site,
    t1.tag_id AS tree1_tag_id,
    s2.name AS tree2_site,
    t2.tag_id AS tree2_tag_id,
    round(st_distance(t1.geog, t2.geog)) AS distance_in_meters,
    round((st_distance(t1.geog, t2.geog) * 3.280839895::double precision)::numeric, 1) AS distance_in_feet
   FROM tree t1
     JOIN tree t2 ON t1.id <> t2.id
     JOIN site s1 ON t1.site_id = s1.id
     JOIN site s2 ON t2.site_id = s2.id
  WHERE st_dwithin(t1.geog, t2.geog, 15::double precision) AND st_distance(t1.geog, t2.geog) <> 0::double precision AND t1.id <> t2.id
  ORDER BY s1.name, s2.name, t1.tag_id, t2.tag_id;

COMMENT ON VIEW public.data_distance_trees_within_15_meters_of_each_other IS 'Show tree to tree distance (in meters and feet) for all trees within 15 meters of each other';
