-- public.data_error_trees_with_gps_approximated source

CREATE OR REPLACE VIEW public.data_error_trees_with_gps_approximated
AS SELECT s.name AS site,
    t.tag_id,
    t.longitude,
    t.latitude,
    t.geog AS gis_map_view,
    t.gps_locked
   FROM tree t
     JOIN site s ON t.site_id = s.id
  WHERE t.gps_locked = false AND t.longitude IS NOT NULL AND t.latitude IS NOT NULL
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.data_error_trees_with_gps_approximated IS 'Trees with estimated longitude and latitude data';
