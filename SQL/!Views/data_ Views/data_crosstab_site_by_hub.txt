-- public.data_crosstab_site_by_hub source

CREATE OR REPLACE VIEW public.data_crosstab_site_by_hub
AS SELECT COALESCE(site, 'Column Total'::text) AS "Site",
    count(*) FILTER (WHERE hub = 'Bethel'::text) AS "Bethel Hub",
    count(*) FILTER (WHERE hub = 'Bridgton'::text) AS "Bridgton Hub",
    count(*) FILTER (WHERE hub = 'Ellsworth'::text) AS "Ellsworth Hub",
    count(*) FILTER (WHERE hub = 'Freeport'::text) AS "Freeport Hub",
    count(*) FILTER (WHERE hub = 'Hartland'::text) AS "Hartland Hub",
    count(*) FILTER (WHERE hub = 'Lincoln'::text) AS "Lincoln Hub",
    count(*) FILTER (WHERE hub = 'Manchester NH'::text) AS "Manchester NH Hub",
    count(*) FILTER (WHERE hub = 'Orono'::text) AS "Orono Hub",
    count(*) FILTER (WHERE hub = 'Princeton'::text) AS "Princeton Hub",
    count(*) FILTER (WHERE hub = 'Sanford'::text) AS "Sanford Hub",
    count(*) FILTER (WHERE hub = 'Searsport'::text) AS "Searsport Hub",
    count(*) FILTER (WHERE hub = 'Warren'::text) AS "Warren Hub",
    count(*) FILTER (WHERE hub = NULL::text) AS "Unknown Hub",
    count(*) AS "Row Total"
   FROM cam_sites_hubs
  GROUP BY ROLLUP(site)
  ORDER BY site;

COMMENT ON VIEW public.data_crosstab_site_by_hub IS 'Crosstab of Site BY hub';

-- Permissions
