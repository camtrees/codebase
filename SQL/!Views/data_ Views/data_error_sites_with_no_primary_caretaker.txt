--DROP VIEW IF EXISTS data_error_sites_with_no_primary_caretaker;

-- public.data_error_sites_with_no_primary_caretaker source

CREATE OR REPLACE VIEW public.data_error_sites_with_no_primary_caretaker
AS SELECT name AS site,
    primary_caretaker_id AS primary_caretaker
   FROM site
  WHERE primary_caretaker_id IS NULL AND lower(name) <> 'wild tree unknown to tacf'::text AND lower(name) <> 'wild tree known to tacf'::text
  ORDER BY name;

COMMENT ON VIEW public.data_error_sites_with_no_primary_caretaker IS 'Sites with no primary caretaker';
