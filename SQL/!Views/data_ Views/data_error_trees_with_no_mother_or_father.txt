------------------------------------------------------------------------------------------
-- View: data_error_trees_with_no_mother_or_father
--
-- Purpose: Trees with missing parents
--
-- Create Order:
--		MUST BE AFTER
--          site
--			tree
--			tree_parent_type
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_error_trees_with_no_mother_or_father;
CREATE OR REPLACE VIEW data_error_trees_with_no_mother_or_father AS
	SELECT
		s.name    AS site_name,
		t.number  AS tree_number,
		p1.parent AS mother_tree,
		p2.parent AS father_tree
	FROM
		tree t
	JOIN
		site s
		ON t.site_id = s.id
	LEFT JOIN
		tree_parent_type p1
		ON t.mother_tree_id = p1.id
	LEFT JOIN
		tree_parent_type p2
		ON t.father_tree_id = p2.id
	WHERE
		(
		t.mother_tree_id IS NULL
		OR t.father_tree_id IS NULL
		)
		AND LOWER(name) <> 'wild tree unknown to tacf'
		AND LOWER(name) <> 'wild tree known to tacf'
		
	ORDER BY
		s.name,
		t.number
		;
