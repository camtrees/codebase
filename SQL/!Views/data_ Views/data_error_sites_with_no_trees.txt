--DROP VIEW IF EXISTS data_error_sites_with_no_trees;

-- public.data_error_sites_with_no_trees source

CREATE OR REPLACE VIEW public.data_error_sites_with_no_trees
AS SELECT s.name AS site_name
   FROM site s
     LEFT JOIN tree t ON s.id = t.site_id
  WHERE t.site_id IS NULL
  ORDER BY s.name;

COMMENT ON VIEW public.data_error_sites_with_no_trees IS 'Sites with no trees';

-- Permissions

ALTER TABLE public.data_error_sites_with_no_trees OWNER TO neondb_owner;
GRANT ALL ON TABLE public.data_error_sites_with_no_trees TO neondb_owner;
