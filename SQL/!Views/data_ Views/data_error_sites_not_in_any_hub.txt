--DROP VIEW IF EXISTS data_error_sites_not_in_any_hub;

-- public.data_error_sites_not_in_any_hub source

CREATE OR REPLACE VIEW public.data_error_sites_not_in_any_hub
AS SELECT s.name AS "Sites not in a Hub"
   FROM site s
     LEFT JOIN site_hub sh ON s.id = sh.site_id
  WHERE sh.site_id IS NULL AND lower(s.name) <> 'wild tree unknown to tacf'::text AND lower(s.name) <> 'wild tree known to tacf'::text
  ORDER BY s.name;

COMMENT ON VIEW public.data_error_sites_not_in_any_hub IS 'Sites not linked to any hubs';

-- Permissions
