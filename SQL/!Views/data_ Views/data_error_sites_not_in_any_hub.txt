------------------------------------------------------------------------------------------
-- View: data_error_sites_not_in_any_hub
--
-- Purpose: Sites not linked to any hubs
--
-- Create Order:
--		MUST BE AFTER
--			site
--			site_hub
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_error_sites_not_in_any_hub;
CREATE OR REPLACE VIEW data_error_sites_not_in_any_hub AS
	SELECT
		s.name AS "Sites not in a Hub"
	FROM
		site s
	LEFT JOIN
		site_hub sh
		ON s.id = sh.site_id
	WHERE
		sh.site_id IS NULL
		AND LOWER(s.name) <> 'wild tree unknown to tacf'
		AND LOWER(s.name) <> 'wild tree known to tacf'
	ORDER BY
		s.name
	;
