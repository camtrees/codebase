-- public.data_error_volunteers_with_no_hometown source

CREATE OR REPLACE VIEW public.data_error_volunteers_with_no_hometown
AS SELECT first_name,
    last_name,
    email,
    town_id AS hometown
   FROM volunteer
  WHERE town_id IS NULL
  ORDER BY first_name, last_name;

COMMENT ON VIEW public.data_error_volunteers_with_no_hometown IS 'Volunteers with no home town';

-- Permissions

ALTER TABLE public.data_error_volunteers_with_no_hometown OWNER TO neondb_owner;
GRANT ALL ON TABLE public.data_error_volunteers_with_no_hometown TO neondb_owner;
