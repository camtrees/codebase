--DROP VIEW IF EXISTS data_error_sites_in_multiple_hubs;

-- public.data_error_sites_in_multiple_hubs source

CREATE OR REPLACE VIEW public.data_error_sites_in_multiple_hubs
AS SELECT s.name AS site,
    h.name AS hub
   FROM site s
     JOIN site_hub sh ON s.id = sh.site_id
     JOIN hub h ON sh.hub_id = h.id
  WHERE (s.id IN ( SELECT site_hub.site_id
           FROM site_hub
          GROUP BY site_hub.site_id
         HAVING count(DISTINCT site_hub.hub_id) > 1))
  ORDER BY s.name, h.name;

COMMENT ON VIEW public.data_error_sites_in_multiple_hubs IS 'Sites in more than one hub';

-- Permissions

GRANT ALL ON TABLE public.data_error_sites_in_multiple_hubs TO neondb_owner;
