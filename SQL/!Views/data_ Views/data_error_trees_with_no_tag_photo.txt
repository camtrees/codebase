------------------------------------------------------------------------------------------
-- View: data_error_trees_with_no_tag_photo
--
-- Purpose: Trees with missing tag photo
--
-- Create Order:
--		MUST BE AFTER
--			tree
--			site
--			tree_photo
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_error_trees_with_no_tag_photo;
CREATE OR REPLACE VIEW data_error_trees_with_no_tag_photo AS
	SELECT
		s.name AS site,
		t.number AS tree_number
	FROM
		tree t
	JOIN
		site s
		ON t.site_id = s.id
	LEFT JOIN
		tree_photo tp
		ON (t.id = tp.tree_id AND tp.tag_photo = true)
	WHERE
		tp.id IS NULL 
		AND LOWER(s.name) <> 'wild tree unknown to tacf'
		AND LOWER(s.name) <> 'wild tree known to tacf'
	ORDER BY
		s.name,
		t.number
