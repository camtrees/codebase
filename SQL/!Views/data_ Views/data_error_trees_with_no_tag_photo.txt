--DROP VIEW IF EXISTS data_error_trees_with_no_tag_photo;

-- public.data_error_trees_with_no_tag_photo source

CREATE OR REPLACE VIEW public.data_error_trees_with_no_tag_photo
AS SELECT s.name AS site,
    t.tag_id
   FROM tree t
     JOIN site s ON t.site_id = s.id
     LEFT JOIN tree_photo tp ON t.id = tp.tree_id AND tp.tag_photo = true
  WHERE tp.id IS NULL AND lower(s.name) <> 'wild tree unknown to tacf'::text AND lower(s.name) <> 'wild tree known to tacf'::text
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.data_error_trees_with_no_tag_photo IS 'Trees with no tag photo';

-- Permissions
