-- public.data_error_hubs_with_no_sites source

CREATE OR REPLACE VIEW public.data_error_hubs_with_no_sites
AS SELECT h.name AS "Hubs with no Sites"
   FROM hub h
     LEFT JOIN site_hub sh ON h.id = sh.hub_id
  WHERE sh.hub_id IS NULL
  ORDER BY h.name;

COMMENT ON VIEW public.data_error_hubs_with_no_sites IS 'Hubs not linked to any sites';

-- Permissions

ALTER TABLE public.data_error_hubs_with_no_sites OWNER TO neondb_owner;
GRANT ALL ON TABLE public.data_error_hubs_with_no_sites TO neondb_owner;
