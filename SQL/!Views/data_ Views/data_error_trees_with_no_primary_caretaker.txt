------------------------------------------------------------------------------------------
-- View: data_error_trees_with_no_primary_caretaker
--
-- Purpose: Trees with no primary caretaker
--
-- Create Order:
--		MUST BE AFTER
--			tree
--			site
--			volunteer
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_error_trees_with_no_primary_caretaker;
CREATE OR REPLACE VIEW data_error_trees_with_no_primary_caretaker AS
	SELECT
		s.name                             AS site,
		t.number                           AS tree_number,
		v.first_name || ' ' || v.last_name AS primary_caretaker
	FROM
		tree t
	JOIN
		site s
		ON t.site_id = s.id
	LEFT JOIN
		volunteer v
		ON t.primary_caretaker_id = v.id
	WHERE
		t.primary_caretaker_id IS NULL
		AND LOWER(s.name) <> 'wild tree unknown to tacf'
		AND LOWER(s.name) <> 'wild tree known to tacf'
	ORDER BY
		s.name,
		t.number
	;
