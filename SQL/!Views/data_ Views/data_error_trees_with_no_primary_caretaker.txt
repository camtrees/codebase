-- public.data_error_trees_with_no_primary_caretaker source

CREATE OR REPLACE VIEW public.data_error_trees_with_no_primary_caretaker
AS SELECT s.name AS site,
    t.tag_id,
    (v.first_name || ' '::text) || v.last_name AS primary_caretaker
   FROM tree t
     JOIN site s ON t.site_id = s.id
     LEFT JOIN volunteer v ON t.primary_caretaker_id = v.id
  WHERE t.primary_caretaker_id IS NULL AND lower(s.name) <> 'wild tree unknown to tacf'::text AND lower(s.name) <> 'wild tree known to tacf'::text
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.data_error_trees_with_no_primary_caretaker IS 'Trees with no primary caretaker';
