--DROP VIEW IF EXISTS data_error_trees_with_no_planting_method;

-- public.data_error_trees_with_no_planting_method source

CREATE OR REPLACE VIEW public.data_error_trees_with_no_planting_method
AS SELECT s.name AS site,
    t.tag_id
   FROM tree t
     JOIN site s ON t.site_id = s.id
  WHERE t.planting_method_id IS NULL AND s.name !~~ 'Wild%'::text
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.data_error_trees_with_no_planting_method IS 'Trees with no planting method';

-- Permissions

ALTER TABLE public.data_error_trees_with_no_planting_method OWNER TO neondb_owner;
GRANT ALL ON TABLE public.data_error_trees_with_no_planting_method TO neondb_owner;
