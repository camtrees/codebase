------------------------------------------------------------------------------------------
-- View: data_duplicate_tree_care_actions
--
-- Purpose: Duplicate tree_care_action records
--
-- Create Order:
--		MUST BE AFTER
--			tree_care_action
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_duplicate_tree_care_actions

-- public.data_duplicate_tree_care_actions source

CREATE OR REPLACE VIEW public.data_duplicate_tree_care_actions
AS SELECT tree_id,
    care_action_id,
    date,
    count(*) AS duplicate_count
   FROM tree_care_action
  GROUP BY tree_id, care_action_id, date
 HAVING count(*) > 1
  ORDER BY tree_id, care_action_id, date;

COMMENT ON VIEW public.data_duplicate_tree_care_actions IS 'Duplicate tree_care_action records (if any)';

-- Permissions

ALTER TABLE public.data_duplicate_tree_care_actions OWNER TO neondb_owner;
GRANT ALL ON TABLE public.data_duplicate_tree_care_actions TO neondb_owner;
