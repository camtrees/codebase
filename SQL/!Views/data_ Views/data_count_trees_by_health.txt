------------------------------------------------------------------------------------------
-- View: data_count_trees_by_health
--
-- Purpose: Show count of trees by the tree health
--
-- Create Order:
--		MUST BE AFTER
--			cam_trees
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_count_trees_by_health;

-- public.data_count_trees_by_health source

CREATE OR REPLACE VIEW public.data_count_trees_by_health
AS SELECT COALESCE(tht.health, 'unknown health'::text) AS "Health",
    count(*) AS "# Trees"
   FROM tree t
     LEFT JOIN tree_health_assessment tha ON t.id = tha.tree_id
     LEFT JOIN tree_health_type tht ON tha.health_id = tht.id
     LEFT JOIN ( SELECT tree_health_assessment.tree_id,
            max(tree_health_assessment.date) AS latest_date
           FROM tree_health_assessment
          WHERE tree_health_assessment.health_id IS NOT NULL
          GROUP BY tree_health_assessment.tree_id) latest_health ON t.id = latest_health.tree_id AND tha.date = latest_health.latest_date
  WHERE tha.id IS NULL OR tha.date = latest_health.latest_date
  GROUP BY (COALESCE(tht.health, 'unknown health'::text))
  ORDER BY (
        CASE COALESCE(tht.health, 'unknown health'::text)
            WHEN 'good'::text THEN 1
            WHEN 'poor'::text THEN 2
            WHEN 'dead'::text THEN 3
            WHEN 'dormant'::text THEN 4
            ELSE 5
        END);

COMMENT ON VIEW public.data_count_trees_by_health IS 'Frequency count of trees by the tree health';
