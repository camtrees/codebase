------------------------------------------------------------------------------------------
-- View: data_distance_between_all_trees
--
-- Purpose: Show tree to tree distance for all trees
--
-- Create Order :
--		MUST BE AFTER
--			site
--			tree
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_distance_between_all_trees;
CREATE OR REPLACE VIEW data_distance_between_all_trees AS
	SELECT
		s1.name                                                             AS tree1_site,
		t1.number                                                           AS tree1_tree_number,
		s2.name                                                             AS tree2_site,
		t2.number                                                           AS tree2_tree_number,		
		ROUND(ST_Distance(t1.geog, t2.geog))                                AS distance_in_meters,
		ROUND((ST_Distance(t1.geog, t2.geog) * 0.0006213711922)::numeric,1) AS distance_in_miles
	FROM
		tree AS t1
	JOIN
		tree AS t2
		ON t1.id <> t2.id -- yields double rows versus using just '<'
	JOIN
		site s1
		ON t1.site_id = s1.id
	JOIN
		site s2
		ON t2.site_id = s2.id
	WHERE
		-- when a site has the same longitude and latitude for all trees at the site
		-- exclude the distance between trees at that site (distance will be 0)
		ST_Distance(t1.geog, t2.geog) <> 0
	ORDER BY
		s1.name,
		s2.name,
		t1.number,
		t2.number
		;
