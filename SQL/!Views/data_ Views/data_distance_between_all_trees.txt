------------------------------------------------------------------------------------------
-- View: data_distance_between_all_trees
--
-- Purpose: Show tree to tree distance for all trees
--
-- Create Order :
--		MUST BE AFTER
--			site
--			tree
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS data_distance_between_all_trees;

-- public.data_distance_between_all_trees source

CREATE OR REPLACE VIEW public.data_distance_between_all_trees
AS SELECT s1.name AS tree1_site,
    t1.tag_id AS tree1_tag_id,
    s2.name AS tree2_site,
    t2.tag_id AS tree2_tag_id,
    round(st_distance(t1.geog, t2.geog)) AS distance_in_meters,
    round((st_distance(t1.geog, t2.geog) * 0.0006213711922::double precision)::numeric, 1) AS distance_in_miles
   FROM tree t1
     JOIN tree t2 ON t1.id <> t2.id
     JOIN site s1 ON t1.site_id = s1.id
     JOIN site s2 ON t2.site_id = s2.id
  WHERE st_distance(t1.geog, t2.geog) <> 0::double precision
  ORDER BY s1.name, s2.name, t1.tag_id, t2.tag_id;

COMMENT ON VIEW public.data_distance_between_all_trees IS 'Show tree to tree distance (in meters and miles) for all trees';
