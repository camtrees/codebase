--DROP VIEW IF EXISTS data_error_trees_with_no_tag_info;

-- public.data_error_trees_with_no_tag_info source

CREATE OR REPLACE VIEW public.data_error_trees_with_no_tag_info
AS SELECT s.name AS site,
    t.tag_id,
    t.tag_info
   FROM tree t
     JOIN site s ON t.site_id = s.id
  WHERE t.tag_info IS NULL AND lower(s.name) <> 'wild tree unknown to tacf'::text AND lower(s.name) <> 'wild tree known to tacf'::text
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.data_error_trees_with_no_tag_info IS 'Trees with no tag_info';

-- Permissions
