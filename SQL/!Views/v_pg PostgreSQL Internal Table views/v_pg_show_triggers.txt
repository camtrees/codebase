--DROP VIEW IF EXISTS v_pg_show_triggers;

-- public.v_pg_show_triggers source

CREATE OR REPLACE VIEW public.v_pg_show_triggers
AS SELECT event_object_table AS table_name,
    trigger_name
   FROM information_schema.triggers
  GROUP BY event_object_table, trigger_name
  ORDER BY event_object_table, trigger_name;

COMMENT ON VIEW public.v_pg_show_triggers IS 'Database Admin View - Show Triggers';

-- Permissions

ALTER TABLE public.v_pg_show_triggers OWNER TO neondb_owner;
GRANT ALL ON TABLE public.v_pg_show_triggers TO neondb_owner;
