-- public.v_pg_show_user_membership source

CREATE OR REPLACE VIEW public.v_pg_show_user_membership
AS SELECT rolname,
    ARRAY( SELECT b.rolname
           FROM pg_auth_members m
             JOIN pg_roles b ON m.roleid = b.oid
          WHERE m.member = r.oid) AS memberof
   FROM pg_roles r
  WHERE rolname <> ALL (ARRAY['pg_signal_backend'::name, 'rds_iam'::name, 'rds_replication'::name, 'rds_superuser'::name, 'rdsadmin'::name, 'rdsrepladmin'::name])
  ORDER BY rolname;

COMMENT ON VIEW public.v_pg_show_user_membership IS 'Database Admin View - Show User membership';

-- Permissions

ALTER TABLE public.v_pg_show_user_membership OWNER TO neondb_owner;
GRANT ALL ON TABLE public.v_pg_show_user_membership TO neondb_owner;
