-- public.v_pg_show_constraints source

CREATE OR REPLACE VIEW public.v_pg_show_constraints
AS SELECT
        CASE
            WHEN pgc.contype = 'p'::"char" THEN 'PKEY'::text
            WHEN pgc.contype = 'f'::"char" THEN 'FKEY'::text
            WHEN pgc.contype = 'n'::"char" THEN 'NOT NULL'::text
            WHEN pgc.contype = 'c'::"char" THEN 'CHECK'::text
            ELSE 'Other'::text
        END AS constraint_type,
    ccu.table_name,
    ccu.column_name,
    pg_get_constraintdef(pgc.oid) AS constraint_definition,
    cc.table_name AS referring_table_name,
    pgc.conname AS constraint_name
   FROM pg_constraint pgc
     JOIN pg_namespace nsp ON nsp.oid = pgc.connamespace
     JOIN pg_class cls ON pgc.conrelid = cls.oid
     LEFT JOIN information_schema.constraint_column_usage ccu ON pgc.conname = ccu.constraint_name::name AND nsp.nspname = ccu.constraint_schema::name
     LEFT JOIN information_schema.table_constraints cc ON pgc.conname = cc.constraint_name::name
  WHERE pgc.connamespace = 'public'::regnamespace::oid AND ccu.table_name::name <> 'NULL'::name
  ORDER BY (
        CASE pgc.contype
            WHEN 'p'::"char" THEN 1
            WHEN 'n'::"char" THEN 2
            WHEN 'c'::"char" THEN 3
            WHEN 'f'::"char" THEN 4
            ELSE 5
        END), ccu.table_name, ccu.column_name, (pg_get_constraintdef(pgc.oid)), pgc.conname;

COMMENT ON VIEW public.v_pg_show_constraints IS 'Database Admin View - Show Constraints';

-- Permissions

ALTER TABLE public.v_pg_show_constraints OWNER TO neondb_owner;
GRANT ALL ON TABLE public.v_pg_show_constraints TO neondb_owner;
