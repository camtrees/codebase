CREATE OR REPLACE VIEW public.v_pg_show_count_table_rows
AS SELECT table_name,
    row_count
   FROM camtrees_count_table_rows() camtrees_count_table_rows(table_name, row_count);

COMMENT ON VIEW public.v_pg_show_count_table_rows IS 'Database Admin View - Show number of rows in all Tables';
