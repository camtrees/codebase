-- public.v_pg_show_functions source

CREATE OR REPLACE VIEW public.v_pg_show_functions
AS SELECT n.nspname AS "Schema",
    u.rolname AS "Owner",
    p.proname AS "Name",
    pg_get_function_result(p.oid) AS "Result data type",
    pg_get_function_arguments(p.oid) AS "Argument data types"
   FROM pg_proc p
     JOIN pg_roles u ON u.oid = p.proowner
     LEFT JOIN pg_namespace n ON n.oid = p.pronamespace
  WHERE pg_function_is_visible(p.oid) AND n.nspname = 'public'::name AND u.rolname = CURRENT_USER;

COMMENT ON VIEW public.v_pg_show_functions IS 'Database Admin View - Show Functions';

-- Permissions

ALTER TABLE public.v_pg_show_functions OWNER TO neondb_owner;
GRANT ALL ON TABLE public.v_pg_show_functions TO neondb_owner;
