------------------------------------------------------------------------------------------
-- View: cam_tree_latest_health
--
-- Purpose: Show latest tree_health
--
-- Create Order:
--		MUST BE AFTER
--			tree_health_assessment
--			tree
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS cam_tree_latest_health;

-- public.cam_tree_latest_health source

CREATE OR REPLACE VIEW public.cam_tree_latest_health
AS WITH ranked_assessments AS (
         SELECT tree_health_assessment.tree_id,
            tree_health_assessment.date,
            tree_health_assessment.health_id,
            row_number() OVER (PARTITION BY tree_health_assessment.tree_id ORDER BY tree_health_assessment.date DESC, tree_health_assessment.id DESC) AS rn
           FROM tree_health_assessment
          WHERE tree_health_assessment.health_id IS NOT NULL
        )
 SELECT s.name AS site,
    t.number AS tree_number,
    ra.date AS latest_health_date,
    tht.health AS latest_health
   FROM tree t
     JOIN site s ON t.site_id = s.id
     JOIN ranked_assessments ra ON t.id = ra.tree_id
     JOIN tree_health_type tht ON ra.health_id = tht.id
  WHERE ra.rn = 1
  ORDER BY s.name, t.number;

COMMENT ON VIEW public.cam_tree_latest_health IS 'Show latest tree_health';

-- Permissions

ALTER TABLE public.cam_tree_latest_health OWNER TO neondb_owner;
GRANT ALL ON TABLE public.cam_tree_latest_health TO neondb_owner;
