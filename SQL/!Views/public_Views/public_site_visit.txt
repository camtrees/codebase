--		MUST BE AFTER
--			cam_tree_latest_health
--			site
--			tree
--			***tree_access_level_type
--			***tree_access_method_type
--			***tree_access_path_type
--			***tree_form_type
--			***tree_planting_method_type

--DROP VIEW IF EXISTS public_site_visit;

-- public.public_site_visit source

CREATE OR REPLACE VIEW public.public_site_visit
AS SELECT s.name AS "Site",
    t.tag_id AS "Tag ID",
    t.date AS "Date Planted",
    ctlh.latest_health AS "Health",
    tapt.access_path AS "Access Path",
    talt.access_level AS "Access Level",
    t.gps_locked AS "GPS Locked",
    tpmt.planting_method AS "Planting Method",
    t.wire_fence AS "Wire Fence",
    -- mother_tree: resolve via tree_tag fallback if tree.mother_tree_id is null
    COALESCE(p1.parent, p1_tag.parent) AS "Mother Tree",
    -- father_tree: resolve via tree_tag fallback if tree.father_tree_id is null
    COALESCE(p2.parent, p2_tag.parent) AS "Father Tree",
    -- parent_tree_note: fall back to tree_tag if null on tree
    COALESCE(t.parent_tree_note, tt.parent_tree_note) AS "Parent Tree Note",
    t.note AS "Note"
   FROM tree t
     LEFT JOIN tree_parent_type p1 ON t.mother_tree_id = p1.id
     LEFT JOIN tree_parent_type p2 ON t.father_tree_id = p2.id
     LEFT JOIN site s ON t.site_id = s.id
     LEFT JOIN tree_access_path_type tapt ON t.access_path_id = tapt.id
     LEFT JOIN tree_access_level_type talt ON t.access_level_id = talt.id
     LEFT JOIN tree_form_type tft ON t.form_id = tft.id
     LEFT JOIN tree_planting_method_type tpmt ON t.planting_method_id = tpmt.id
     -- Join tree_tag ON matching tag_id to provide fallback values
     LEFT JOIN public.tree_tag tt ON (t.tag_id = tt.tag_id)
     -- Resolve tree_tag mother/father parent names for fallback
     LEFT JOIN public.tree_parent_type p1_tag ON (tt.mother_tree_id = p1_tag.id)
     LEFT JOIN public.tree_parent_type p2_tag ON (tt.father_tree_id = p2_tag.id)
     LEFT JOIN cam_tree_latest_health ctlh ON s.name = ctlh.site AND t.tag_id = ctlh.tag_id
  ORDER BY s.name, t.tag_id;

COMMENT ON VIEW public.public_site_visit IS 'Tree data useful to have printed when performing a site visit';
