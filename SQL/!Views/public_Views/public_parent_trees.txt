------------------------------------------------------------------------------------------
-- View: public_parent_trees
--
-- Purpose: All the possible parent trees
------------------------------------------------------------------------------------------
--DROP VIEW IF EXISTS public_parent_trees;
CREATE OR REPLACE VIEW public_parent_trees AS
	SELECT
		parent AS parent_tree
	FROM
		tree_parent_type
	ORDER BY
		parent
	;

COMMENT ON VIEW public.public_parent_trees IS 'A sanitized (no email addresses) view of parent trees';

-- Permissions

ALTER TABLE public.public_parent_trees OWNER TO neondb_owner;
GRANT ALL ON TABLE public.public_parent_trees TO neondb_owner;
